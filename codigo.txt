#include <stdio.h>

#define TAM 10


typedef struct {
    char nome[30];
    float preco;
} Produto;


void mostrarProdutos(Produto produtos[], int tamanho) {
    for (int i = 0; i < tamanho; i++) {
        printf("%-20s R$ %.2f\n",
               produtos[i].nome, produtos[i].preco);
    }
}



void bubbleSort(Produto produtos[], int tamanho) {
    for (int i = 0; i < tamanho - 1; i++) {
        for (int j = 0; j < tamanho - 1 - i; j++) {
            if (produtos[j].preco > produtos[j + 1].preco) {
                Produto temp = produtos[j];
                produtos[j] = produtos[j + 1];
                produtos[j + 1] = temp;
            }
        }
    }
}


int particionar(Produto produtos[], int inicio, int fim) {
    float pivo = produtos[fim].preco;
    int i = inicio - 1;

    for (int j = inicio; j < fim; j++) {
        if (produtos[j].preco <= pivo) {
            i++;

            Produto temp = produtos[i];
            produtos[i] = produtos[j];
            produtos[j] = temp;
        }
    }

    Produto temp = produtos[i + 1];
    produtos[i + 1] = produtos[fim];
    produtos[fim] = temp;

    return i + 1;
}


void quickSort(Produto produtos[], int inicio, int fim) {
    if (inicio < fim) {
        int pivo = particionar(produtos, inicio, fim);

        quickSort(produtos, inicio, pivo - 1);
        quickSort(produtos, pivo + 1, fim);
    }
}

int main() {
    Produto produtos[TAM] = {
        {"Arroz", 28.90},
        {"Feijao", 8.50},
        {"Leite", 5.49},
        {"Cafe", 18.90},
        {"Acucar", 4.79},
        {"Macarrao", 6.99},
        {"Oleo", 7.50},
        {"Farinha", 5.99},
        {"Biscoito", 3.89},
        {"Queijo", 32.90}
    };

    
    Produto bubble[TAM];
    Produto quick[TAM];

    for (int i = 0; i < TAM; i++) {
        bubble[i] = produtos[i];
        quick[i] = produtos[i];
    }

    printf("=== SISTEMA DE PRECOS DO SUPERMERCADO ===\n");

    printf("\nProdutos antes da ordenacao:\n");
    mostrarProdutos(produtos, TAM);

    
    bubbleSort(bubble, TAM);

    printf("\n=== RESULTADO DO BUBBLE SORT ===\n");
    mostrarProdutos(bubble, TAM);

    
    quickSort(quick, 0, TAM - 1);

    printf("\n=== RESULTADO DO QUICK SORT ===\n");
    mostrarProdutos(quick, TAM);

    return 0;
}
